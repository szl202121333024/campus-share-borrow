# app.py —— 添加物品功能（Flask + SQLite，单文件即可运行）
# 安装依赖: pip install flask
# 启动:     python app.py   ->  http://127.0.0.1:5000
import os
import sqlite3
from datetime import datetime

from flask import (Flask, flash, g, redirect, render_template_string,
                   request, url_for)

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
DB_PATH = os.path.join(BASE_DIR, "inventory.db")

app = Flask(__name__)
app.config["SECRET_KEY"] = os.environ.get("SECRET_KEY", "dev-only-secret-key")

CATEGORIES = ("电子设备", "办公用品", "工具", "耗材", "其他")
UNITS = ("个", "台", "件", "箱", "包", "米", "公斤")

MAX_QTY = 1_000_000
MAX_PRICE = 1e9


# ----------------------------- 数据库 -----------------------------
def get_db():
    if "db" not in g:
        g.db = sqlite3.connect(DB_PATH)
        g.db.row_factory = sqlite3.Row
        g.db.execute("PRAGMA foreign_keys = ON")
    return g.db


@app.teardown_appcontext
def close_db(exc):
    db = g.pop("db", None)
    if db is not None:
        db.close()


def init_db():
    with sqlite3.connect(DB_PATH) as db:
        db.executescript(
            """
            CREATE TABLE IF NOT EXISTS items (
                id         INTEGER PRIMARY KEY AUTOINCREMENT,
                name       TEXT    NOT NULL UNIQUE,
                category   TEXT    NOT NULL,
                quantity   INTEGER NOT NULL DEFAULT 0 CHECK (quantity >= 0),
                unit       TEXT    NOT NULL DEFAULT '个',
                price      REAL    NOT NULL DEFAULT 0 CHECK (price >= 0),
                location   TEXT    NOT NULL DEFAULT '',
                note       TEXT    NOT NULL DEFAULT '',
                created_at TEXT    NOT NULL
            );
            CREATE INDEX IF NOT EXISTS idx_items_created_at
                ON items(created_at DESC);
            """
        )


# ----------------------------- 表单校验 -----------------------------
def validate_item(form):
    """返回 (清洗后的数据, 错误字典)；两个字段名互不重叠。"""
    raw = {
        "name": (form.get("name") or "").strip(),
        "category": (form.get("category") or "").strip(),
        "quantity": (form.get("quantity") or "").strip(),
        "unit": (form.get("unit") or "").strip(),
        "price": (form.get("price") or "").strip(),
        "location": (form.get("location") or "").strip(),
        "note": (form.get("note") or "").strip(),
    }
    errors = {}

    # 名称：必填，长度 1~50
    if not raw["name"]:
        errors["name"] = "物品名称不能为空"
    elif len(raw["name"]) > 50:
        errors["name"] = "物品名称不能超过 50 个字符"

    # 分类 / 单位：必须来自白名单，防止脏数据
    if raw["category"] not in CATEGORIES:
        errors["category"] = "请选择有效的分类"
    if raw["unit"] not in UNITS:
        errors["unit"] = "请选择有效的单位"

    # 数量：必填，非负整数
    quantity = None
    if not raw["quantity"]:
        errors["quantity"] = "数量不能为空"
    else:
        try:
            quantity = int(raw["quantity"])
        except ValueError:
            errors["quantity"] = "数量必须是整数"
        else:
            if quantity < 0 or quantity > MAX_QTY:
                errors["quantity"] = f"数量需在 0 ~ {MAX_QTY} 之间"

    # 单价：选填，非负数字
    price = 0.0
    if raw["price"]:
        try:
            price = float(raw["price"])
        except ValueError:
            errors["price"] = "单价必须是数字"
        else:
            if price < 0 or price > MAX_PRICE:
                errors["price"] = "单价必须是 0 ~ 1000000000 之间的数字"

    if len(raw["location"]) > 50:
        errors["location"] = "存放位置不能超过 50 个字符"
    if len(raw["note"]) > 200:
        errors["note"] = "备注不能超过 200 个字符"

    clean = {
        "name": raw["name"],
        "category": raw["category"],
        "quantity": quantity,
        "unit": raw["unit"],
        "price": round(price, 2),
        "location": raw["location"],
        "note": raw["note"],
    }
    return clean, errors


# ----------------------------- 页面渲染 -----------------------------
def render_page(status=200, form=None, errors=None):
    items = get_db().execute(
        "SELECT * FROM items ORDER BY created_at DESC, id DESC"
    ).fetchall()
    return render_template_string(
        PAGE,
        items=items,
        form=form or {},
        errors=errors or {},
        categories=CATEGORIES,
        units=UNITS,
        total_kinds=len(items),
        total_qty=sum(row["quantity"] for row in items),
        total_value=sum(row["quantity"] * row["price"] for row in items),
    ), status


# ----------------------------- 路由 -----------------------------
@app.get("/")
def index():
    return render_page()


@app.post("/items")
def add_item():
    clean, errors = validate_item(request.form)

    if errors:
        # 校验失败：回填用户输入 + 显示错误，返回 400
        return render_page(status=400, form=request.form, errors=errors)

    db = get_db()
    try:
        db.execute(
            """
            INSERT INTO items
                (name, category, quantity, unit, price, location, note, created_at)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
            """,
            (
                clean["name"], clean["category"], clean["quantity"],
                clean["unit"], clean["price"], clean["location"],
                clean["note"], datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            ),
        )
        db.commit()
    except sqlite3.IntegrityError:
        # name 上有 UNIQUE 约束：同名物品直接给出友好提示
        return render_page(
            status=400,
            form=request.form,
            errors={"name": "已存在同名物品，请换一个名称或直接修改数量"},
        )

    flash(f"已添加物品「{clean['name']}」× {clean['quantity']}{clean['unit']}", "success")
    return redirect(url_for("index"))   # PRG：重定向，刷新不会重复提交
